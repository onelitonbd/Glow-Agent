And here's the function that produces that grouping:

python 
# group and count by status
def status_breakdown(rows):
    counts = {}
    for row in rows:
        key = row["status"]
        counts[key] = counts.get(key, 0) + 1
    return sorted(counts.items(), key=lambda x: -x[1])
